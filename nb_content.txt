## Dataset acquisition

The learning set contains the 35 labeled records used by the pipeline. The official challenge test records (`x01`–`x35`) are not used because they do not provide reference labels for this supervised evaluation.

The downloader below skips already-complete files, writes through a temporary `.part` file, retries interrupted downloads, and the next integrity step verifies that the resulting WFDB records can actually be read.

# STEP 4 — ROBUST APNEA-ECG DOWNLOAD
# ============================================================

from concurrent.futures import ThreadPoolExecutor, as_completed
from tqdm.auto import tqdm
import time

REQUIRED_EXTENSIONS = [".dat", ".hea", ".apn", ".qrs"]

session = requests.Session()
session.headers.update({
    "User-Agent": "Sleep-Apnea-Research-Notebook/1.0"
})

def download_file(url, destination, retries=3, timeout=(15, 120)):
    destination = Path(destination)
    destination.parent.mkdir(parents=True, exist_ok=True)

    if destination.exists() and destination.stat().st_size > 0:
        return {"path": str(destination), "status": "already_exists"}

    part_path = Path(str(destination) + ".part")

    last_error = None

    for attempt in range(1, retries + 1):
        try:
            with session.get(url, stream=True, timeout=timeout) as response:
                response.raise_for_status()

                with open(part_path, "wb") as f:
                    for chunk in response.iter_content(chunk_size=1024 * 1024):
                        if chunk:
                            f.write(chunk)

            if part_path.stat().st_size == 0:
                raise IOError("Downloaded file is empty.")

            part_path.replace(destination)

            return {
                "path": str(destination),
                "status": "downloaded",
                "bytes": destination.stat().st_size
            }

        except Exception as exc:
            last_error = exc
            if part_path.exists():
                part_path.unlink(missing_ok=True)
            time.sleep(attempt)

    raise RuntimeError(f"Failed to download {url}: {last_error}")

def download_record(record_name):
    results = []
    for ext in REQUIRED_EXTENSIONS:
        filename = f"{record_name}{ext}"
        destination = DATA_DIR / filename
        url = PHYSIONET_BASE_URL + filename
        results.append(download_file(url, destination))
    return record_name, results

download_jobs = {}
for record_name in LEARNING_RECORDS:
    for ext in REQUIRED_EXTENSIONS:
        download_jobs[(record_name, ext)] = DATA_DIR / f"{record_name}{ext}"

missing_before = sum(
    not (path.exists() and path.stat().st_size > 0)
    for path in download_jobs.values()
)

print("Missing/non-empty-check failures before download:", missing_before)

with ThreadPoolExecutor(max_workers=6) as executor:

        executor.submit(download_record, rec): rec
        for rec in LEARNING_RECORDS

## Feature engineering

The feature extraction follows the supplied notebooks: each apnea annotation is treated as a 60-second window, RR intervals are filtered to 0.30–2.00 seconds, and time-domain HR/HRV features are computed from the usable RR intervals.

// ==================== CELL 9 [code] ====================
# ============================================================
# STEP 6 — MINUTE-LEVEL HR/HRV FEATURE EXTRACTION
# ============================================================

import json
import warnings

warnings.filterwarnings("ignore")

FS_EXPECTED = 100
MIN_RR = 0.30
MAX_RR = 2.00
MIN_VALID_RR = 3

def extract_hrv_features(r_peaks, start_sample, end_sample, fs=100):
    peaks = r_peaks[
        (r_peaks >= start_sample) &
        (r_peaks < end_sample)
    ]

    if len(peaks) < 4:
        return None

    rr_all = np.diff(peaks) / fs

    if len(rr_all) == 0:
        return None

    rr = rr_all[
        (rr_all >= MIN_RR) &
        (rr_all <= MAX_RR)
    ]

    if len(rr) < MIN_VALID_RR:
        return None

    rr_valid_ratio = len(rr) / len(rr_all)
    hr = 60.0 / rr
    diff_rr = np.diff(rr)

    if len(diff_rr) == 0:
        return None

    mean_hr = np.mean(hr)
    median_hr = np.median(hr)
    std_hr = np.std(hr)
    min_hr = np.min(hr)
    max_hr = np.max(hr)

    mean_rr = np.mean(rr)
    median_rr = np.median(rr)

    sdnn = np.std(rr, ddof=1)
    rmssd = np.sqrt(np.mean(diff_rr ** 2))
    pnn50 = np.mean(np.abs(diff_rr) > 0.05) * 100.0

    rr_iqr = np.percentile(rr, 75) - np.percentile(rr, 25)
    rr_range = np.max(rr) - np.min(rr)
    rr_std = np.std(rr)
    rr_cv = rr_std / mean_rr if mean_rr > 0 else np.nan

    rr_median = np.median(rr)
    rr_mad = np.median(np.abs(rr - rr_median))

    return {
        "mean_hr": mean_hr,
        "median_hr": median_hr,
        "std_hr": std_hr,
        "min_hr": min_hr,
        "max_hr": max_hr,
        "mean_rr": mean_rr,
        "median_rr": median_rr,
        "sdnn": sdnn,
        "rmssd": rmssd,
        "pnn50": pnn50,
        "rr_iqr": rr_iqr,
        "rr_range": rr_range,
        "rr_std": rr_std,
        "rr_cv": rr_cv,
        "rr_mad": rr_mad,
        "num_r_peaks": len(peaks),
        "num_valid_rr": len(rr),
        "rr_valid_ratio": rr_valid_ratio,
    }

all_features = []
record_summaries = []

for record_idx, record_name in enumerate(LEARNING_RECORDS, start=1):
    record_path = str(DATA_DIR / record_name)

    header = wfdb.rdheader(record_path)
    fs = header.fs
    total_samples = header.sig_len

    qrs = wfdb.rdann(record_path, extension="qrs")
    r_peaks = np.asarray(qrs.sample, dtype=np.int64)

    apn = wfdb.rdann(record_path, extension="apn")
    apnea_samples = np.asarray(apn.sample, dtype=np.int64)
    apnea_symbols = np.asarray(apn.symbol)

    successful_minutes = 0
    failed_minutes = 0

    for sample, label in zip(apnea_samples, apnea_symbols):
        start_sample = int(sample)
        end_sample = min(start_sample + 60 * int(fs), total_samples)

        features = extract_hrv_features(
            r_peaks=r_peaks,
            start_sample=start_sample,
            end_sample=end_sample,
            fs=fs,
        )

        if features is None:
            failed_minutes += 1
            continue

        minute_number = start_sample // int(fs * 60)

        row = {
            "record": record_name,
            "minute": minute_number,
            "start_sample": start_sample,
            "end_sample": end_sample,
            "label": label,
        }
        row.update(features)
        all_features.append(row)
        successful_minutes += 1

    record_summaries.append({
        "record": record_name,
        "fs": fs,
        "total_samples": total_samples,
        "duration_hours": total_samples / fs / 3600,
        "apnea_annotations": len(apnea_samples),
        "successful_minutes": successful_minutes,
        "failed_minutes": failed_minutes,
        "qrs_peaks": len(r_peaks),
        "status": "OK",
    })

    print(
        f"[{record_idx:02d}/{len(LEARNING_RECORDS)}] "
        f"{record_name}: usable={successful_minutes}, failed={failed_minutes}"
    )

from sklearn.model_selection import train_test_split

record_stats = (
    df.groupby("record")
      .agg(
          total_minutes=("label", "size"),
          apnea_minutes=("label", "sum")
      )
      .reset_index()

record_stats["apnea_ratio"] = (
    record_stats["apnea_minutes"] / record_stats["total_minutes"]
)

record_stats["group"] = record_stats["record"]
record_stats.loc[
    record_stats["record"].isin(["c05", "c06"]),
    "group"
] = "c05_c06"

group_stats = (
    record_stats.groupby("group")
                .agg(
                    total_minutes=("total_minutes", "sum"),
                    apnea_minutes=("apnea_minutes", "sum")
                )
                .reset_index()
)

group_stats["apnea_ratio"] = (
    group_stats["apnea_minutes"] / group_stats["total_minutes"]
)

groups = group_stats["group"].to_numpy()

# Stratify independent groups by apnea burden.
group_stats["stratification_bin"] = pd.qcut(
    group_stats["apnea_ratio"],
    q=4,
    labels=False,
    duplicates="drop"
)

try:
    train_groups, temp_groups = train_test_split(
        groups,
        test_size=0.30,
        random_state=SEED,
        stratify=group_stats["stratification_bin"]
    )

    temp_stats = group_stats[group_stats["group"].isin(temp_groups)].copy()

    val_groups, test_groups = train_test_split(
        temp_groups,
        test_size=0.50,
        random_state=SEED,
        stratify=temp_stats["stratification_bin"]
    )
except ValueError:
    # Deterministic fallback if stratification is impossible for a tiny bin.
    rng = np.random.default_rng(SEED)
    shuffled = groups.copy()
    rng.shuffle(shuffled)
    n_train = int(round(0.70 * len(shuffled)))
    train_groups = shuffled[:n_train]
    temp_groups = shuffled[n_train:]
    midpoint = len(temp_groups) // 2
    val_groups = temp_groups[:midpoint]
    test_groups = temp_groups[midpoint:]

train_groups = sorted(train_groups.tolist())
val_groups = sorted(val_groups.tolist())
test_groups = sorted(test_groups.tolist())

def expand_groups(group_list):
    expanded = []
    for group in group_list:
        if group == "c05_c06":
            expanded.extend(["c05", "c06"])
        else:

import time

AE_INPUT_DIM = X_train_selected.shape[1]
AE_LATENT_DIM = 4
AE_HIDDEN_DIM = 8
AE_BATCH_SIZE = 128
AE_EPOCHS = 200
AE_LEARNING_RATE = 1e-3
AE_WEIGHT_DECAY = 1e-5
AE_PATIENCE = 20

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

class HRVAutoencoder(nn.Module):
    def __init__(self, input_dim=10, hidden_dim=8, latent_dim=4):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, latent_dim),
        )
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, input_dim),
        )

    def forward(self, x):
        z = self.encoder(x)
        reconstructed = self.decoder(z)
        return reconstructed, z

train_tensor = torch.tensor(X_train_selected, dtype=torch.float32)
generator = torch.Generator()
generator.manual_seed(SEED)

train_loader = DataLoader(
    TensorDataset(train_tensor),
    batch_size=AE_BATCH_SIZE,
    shuffle=True,
    generator=generator,
)

ae = HRVAutoencoder(
    input_dim=AE_INPUT_DIM,
    hidden_dim=AE_HIDDEN_DIM,
    latent_dim=AE_LATENT_DIM,
).to(device)

criterion = nn.MSELoss()

optimizer = torch.optim.Adam(
    ae.parameters(),
    lr=AE_LEARNING_RATE,
    weight_decay=AE_WEIGHT_DECAY,
)

best_loss = np.inf
best_state = None
patience_counter = 0
loss_history = []

start = time.perf_counter()

for epoch in range(AE_EPOCHS):
    ae.train()

    epoch_loss = 0.0
    n_seen = 0

    for (batch_x,) in train_loader:
        batch_x = batch_x.to(device)

        optimizer.zero_grad()
        reconstructed, _ = ae(batch_x)

        loss = criterion(reconstructed, batch_x)
        loss.backward()
        optimizer.step()

        batch_size = batch_x.size(0)
        epoch_loss += loss.item() * batch_size
        n_seen += batch_size

    epoch_loss /= n_seen
    loss_history.append(epoch_loss)

    if epoch_loss < best_loss:
        best_loss = epoch_loss
        best_state = {
            key: value.detach().cpu().clone()
            for key, value in ae.state_dict().items()
        }
        patience_counter = 0
    else:
        patience_counter += 1

    if epoch == 0 or (epoch + 1) % 20 == 0:
        print(
            f"Epoch {epoch + 1:3d}/{AE_EPOCHS} | "
            f"train reconstruction MSE={epoch_loss:.7f}"
        )

    if patience_counter >= AE_PATIENCE:
        print("Early stopping at epoch", epoch + 1)
        break

if best_state is None:
    raise RuntimeError("Autoencoder training failed to produce a valid state.")

ae.load_state_dict(best_state)
ae.eval()

print("\nDevice:", device)
print("Best train reconstruction MSE:", best_loss)
print("Epochs completed:", len(loss_history))
print("Training time (sec):", round(time.perf_counter() - start, 2))

from sklearn.svm import SVC
from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    f1_score,
    roc_auc_score,
    average_precision_score,
    confusion_matrix,
    roc_curve,
    brier_score_loss,
    log_loss,
)

def evaluate_scores(y_true, scores, split_name, threshold=0.0, model_name="model", representation=""):
    y_true = np.asarray(y_true).astype(int)
    scores = np.asarray(scores).ravel()
    y_pred = (scores >= threshold).astype(int)

    tn, fp, fn, tp = confusion_matrix(
        y_true, y_pred, labels=[0, 1]
    ).ravel()

    return {
        "model": model_name,
        "representation": representation,
        "split": split_name,
        "threshold": float(threshold),
        "accuracy": accuracy_score(y_true, y_pred),
        "precision": precision_score(y_true, y_pred, zero_division=0),
        "sensitivity": recall_score(y_true, y_pred, zero_division=0),
        "specificity": tn / (tn + fp) if (tn + fp) else 0.0,
        "f1": f1_score(y_true, y_pred, zero_division=0),
        "roc_auc": roc_auc_score(y_true, scores),
        "pr_auc": average_precision_score(y_true, scores),
        "tn": int(tn),
        "fp": int(fp),
        "fn": int(fn),
        "tp": int(tp),
    }

print("Evaluation helpers created.")

def record_balanced_indices(y, records, n_per_class=500, seed=SEED):
    y = np.asarray(y)
    records = np.asarray(records)

    rng = np.random.default_rng(seed)
    selected = []

    for cls in [0, 1]:
        cls_records = np.unique(records[y == cls])
        cls_records = cls_records.copy()
        rng.shuffle(cls_records)

        pools = {
            rec: rng.permutation(
                np.where((y == cls) & (records == rec))[0]
            )
            for rec in cls_records
        }

        chosen = []

        while len(chosen) < n_per_class and any(p.size for p in pools.values()):
            for rec in cls_records:
                if pools[rec].size and len(chosen) < n_per_class:
                    chosen.append(int(pools[rec][0]))
                    pools[rec] = pools[rec][1:]

        if len(chosen) < n_per_class:
            raise ValueError(
                f"Not enough class-{cls} samples for requested balanced subset."
            )

        selected.extend(chosen)

    selected = np.asarray(selected, dtype=int)
    rng.shuffle(selected)

    return selected

train_sample_records = train_df["record"].to_numpy()
val_sample_records = val_df["record"].to_numpy()
test_sample_records = test_df["record"].to_numpy()

QSVM_TRAIN_PER_CLASS = 500

qs_train_idx = record_balanced_indices(
    y_train,
    train_sample_records,
    n_per_class=QSVM_TRAIN_PER_CLASS,
    seed=SEED
)

X_q_train = X_train_ae_scaled[qs_train_idx]

## Quantum pipeline

Two complementary ideas from the supplied notebooks are retained:

1. **Qiskit FidelityQuantumKernel sanity check** on a tiny matrix, so the quantum environment and kernel object are verified before the larger experiment.
2. **Manual Statevector fidelity kernel** for the actual 1,000-sample experiment, which avoids repeated high-level kernel evaluation overhead and makes the matrix orientation explicit for scikit-learn's `precomputed` SVM.

The autoencoder latent vectors are additionally mapped to `[0, π]` using a scaler fitted on TRAIN only. This addresses the scaling problem identified in the supplied comparison while preserving the train-only preprocessing discipline.

def statevectors_from_features(X, fmap, desc="Quantum states"):
    X = np.asarray(X)
    dim = 2 ** fmap.num_qubits

    out = np.empty(
        (len(X), dim),
        dtype=np.complex128
    )

    for i, x in enumerate(tqdm(X, total=len(X), desc=desc)):
        circuit = fmap.assign_parameters(x)
        out[i] = Statevector.from_instruction(circuit).data

    return out

def fidelity_kernel_from_states(A, B):
    return np.abs(A.conj() @ B.T) ** 2

## Explainability

The explanation method is deliberately simple and reproducible: perturb one selected HR/HRV feature at a time and measure the absolute change in the final QSVM decision score. This is a sensitivity analysis, not a causal claim.

// ==================== CELL 41 [code] ====================
# ============================================================
# STEP 34 — SINGLE-SAMPLE QSVM PREDICTION + EXPLAINABILITY
# ============================================================

def predict_sleep_apnea_qsvm(
    raw_selected_features,
    feature_map,
    autoencoder,
    original_scaler,
    latent_scaler,
    quantum_scaler,
    trained_qsvm,
    train_quantum_states
):
    raw = np.asarray(raw_selected_features, dtype=np.float64).reshape(1, -1)

    if raw.shape[1] != len(SELECTED_FEATURES):
        raise ValueError(
            f"Expected {len(SELECTED_FEATURES)} selected features, "
            f"got {raw.shape[1]}."
        )

    raw_df = pd.DataFrame(raw, columns=SELECTED_FEATURES)

    # Fill using medians learned from TRAIN only.
    raw_df = raw_df.fillna(
        pd.Series(
            train_medians[SELECTED_FEATURES].to_dict()
        )
    )

    scaled_original = original_scaler.transform(raw_df).astype(np.float32)

    tensor_x = torch.tensor(
        scaled_original,
        dtype=torch.float32,
        device=device
    )

    autoencoder.eval()
    with torch.no_grad():
        latent = autoencoder.encoder(tensor_x).cpu().numpy().astype(np.float32)

    latent_scaled = latent_scaler.transform(latent).astype(np.float32)
    quantum_x = quantum_scaler.transform(latent_scaled)
    quantum_x = np.clip(quantum_x, 0.0, np.pi).astype(np.float32)

    state = Statevector.from_instruction(
        feature_map.assign_parameters(quantum_x[0])
    ).data.reshape(1, -1)

    kernel_row = fidelity_kernel_from_states(
        state,
        train_quantum_states
    )

    decision_score = float(
        trained_qsvm.decision_function(kernel_row).ravel()[0]
    )

    calibrated_probability = float(
        calibrator.predict_proba(
            np.array([[decision_score]])
        )[0, 1]
    )

    prediction = int(decision_score >= QSVM_THRESHOLD_YOUDEN)

    return {
        "prediction": prediction,
        "decision_score": decision_score,
        "calibrated_probability": calibrated_probability,
        "latent_representation": latent[0],
        "quantum_features": quantum_x[0],
    }

# STEP 35 — PERTURBATION-BASED QSVM EXPLAINABILITY

def qsvm_decision_from_raw_features(raw_features):
    result = predict_sleep_apnea_qsvm(
        raw_selected_features=raw_features,
        feature_map=feature_map,
        autoencoder=ae,
        original_scaler=scaler_selected,
        latent_scaler=ae_latent_scaler,
        quantum_scaler=quantum_angle_scaler,
        trained_qsvm=final_qsvm,
        train_quantum_states=train_states,
    )
    return result["decision_score"]

baseline_raw = demo_raw.copy()
baseline_score = qsvm_decision_from_raw_features(baseline_raw)

feature_scale = scaler_selected.scale_

explanation_rows = []

for i, feature in enumerate(SELECTED_FEATURES):
    perturbed = baseline_raw.copy()

    delta = 0.10 * feature_scale[i]
    if delta == 0:
        delta = 0.10

    perturbed[i] += delta

    perturbed_score = qsvm_decision_from_raw_features(perturbed)
    influence = perturbed_score - baseline_score

    explanation_rows.append({
        "feature": feature,
        "perturbation": delta,
        "baseline_score": baseline_score,
        "perturbed_score": perturbed_score,
        "influence": influence,
        "absolute_influence": abs(influence),
    })

explainability_df = (
    pd.DataFrame(explanation_rows)
      .sort_values("absolute_influence", ascending=False)
      .reset_index(drop=True)
)

display(explainability_df.round(6))

explainability_df.to_csv(
    RESULT_DIR / "qsvm_feature_perturbation_explainability.csv",
    index=False
)
